-- ==========================================
-- 1. テーブル作成・修正セクション
-- ==========================================

-- profilesテーブル（すでに存在していれば足りない部分だけ補完されます）
CREATE TABLE IF NOT EXISTS public.profiles (
    id uuid NOT NULL PRIMARY KEY,
    name text NOT NULL,
    avatar_url text,
    birthday date,
    location text,
    occupation text,
    hobbies text,
    comment text,
    bio text,
    liquor_type text,
    favorite_brand text,
    drinking_style text,
    drinking_area text,
    recommended_bar text,
    bar_url text,
    bar_image_url text,
    status text NOT NULL DEFAULT 'pending',
    role text NOT NULL DEFAULT 'member'
);

-- 通常イベントテーブル (events) の作成・修正
CREATE TABLE IF NOT EXISTS public.events (
    id uuid NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
    event_date text NOT NULL,
    event_timestamp timestamp with time zone,
    title text NOT NULL,
    location text,
    description text,
    created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
    created_by uuid REFERENCES public.profiles(id),
    url text,
    color_class text DEFAULT 'bg-indigo-600'::text,
    deleted_at timestamp with time zone,
    genre text DEFAULT '飲み会'::text,
    participants_count integer DEFAULT 1,
    details text,
    participant_ids text[] DEFAULT '{}'::text[],
    image_url text
);

-- 日常の呟きテーブル (posts) 
CREATE TABLE public.posts (
    id uuid NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
    user_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    content text NOT NULL,
    image_url text,
    created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now())
);

-- 呟きへの返信テーブル (post_replies) 
CREATE TABLE public.post_replies (
    id uuid NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
    post_id uuid NOT NULL REFERENCES public.posts(id) ON DELETE CASCADE,
    user_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    content text NOT NULL,
    created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now())
);

-- いいね（星・ハート）テーブル (post_likes)
CREATE TABLE public.post_likes (
    post_id uuid NOT NULL REFERENCES public.posts(id) ON DELETE CASCADE,
    user_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
    PRIMARY KEY (post_id, user_id)
);

-- チャットルームテーブル
CREATE TABLE public.chat_rooms (
    id uuid NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
    title text NOT NULL,
    created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
    created_by uuid REFERENCES public.profiles(id),
    source_post_id uuid REFERENCES public.posts(id) ON DELETE CASCADE
);

-- チャットメッセージテーブル
CREATE TABLE public.chat_messages (
    id uuid NOT NULL DEFAULT gen_random_uuid() PRIMARY KEY,
    room_id uuid NOT NULL REFERENCES public.chat_rooms(id) ON DELETE CASCADE,
    user_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    message text NOT NULL,
    image_url text,
    created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
    read_user_ids text[] DEFAULT '{}'::text[]
);

-- 既存の古いポリシーを一度削除
DROP POLICY IF EXISTS "Allow approved users or self" ON public.profiles;
DROP POLICY IF EXISTS "Allow approved users only for events" ON public.events;
DROP POLICY IF EXISTS "Allow approved users only for posts" ON public.posts;
DROP POLICY IF EXISTS "Allow approved users only for replies" ON public.post_replies;
DROP POLICY IF EXISTS "Allow approved users only for likes" ON public.post_likes;
DROP POLICY IF EXISTS "Allow approved users only for chat_rooms" ON public.chat_rooms;
DROP POLICY IF EXISTS "Allow approved users only for chat_messages" ON public.chat_messages;

-- ポリシーを再作成（これでstatusの参照エラーが消えます！）
CREATE POLICY "Allow approved users or self" ON public.profiles
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved' OR auth.uid() = id);

CREATE POLICY "Allow approved users only for events" ON public.events
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved');

CREATE POLICY "Allow approved users only for posts" ON public.posts
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved');

CREATE POLICY "Allow approved users only for replies" ON public.post_replies
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved');

CREATE POLICY "Allow approved users only for likes" ON public.post_likes
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved');

CREATE POLICY "Allow approved users only for chat_rooms" ON public.chat_rooms
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved');

CREATE POLICY "Allow approved users only for chat_messages" ON public.chat_messages
    FOR ALL USING ((SELECT status FROM public.profiles WHERE id = auth.uid()) = 'approved');
